    // ---------- MODALES ----------
    function openModal(id){
      const el = document.getElementById(id);
      if(!el) return;
      el.style.display = 'flex';
      el.setAttribute('aria-hidden','false');
      const modal = el.querySelector('.modal');
      setTimeout(()=> modal.classList.add('modal-show'), 10);
      // set date fields
      const today = new Date().toISOString().slice(0,10);
      el.querySelectorAll('input[name="fecha"]').forEach(i=>i.value = today);
    }
    function closeModal(id){
      const el = document.getElementById(id);
      if(!el) return;
      const modal = el.querySelector('.modal');
      if(modal) modal.classList.remove('modal-show');
      setTimeout(()=> { el.style.display = 'none'; el.setAttribute('aria-hidden','true'); }, 240);
    }
    function scrollToTopAndClose(id){
      window.scrollTo({top:0,behavior:'smooth'});
      setTimeout(()=> closeModal(id), 260);
    }
    document.getElementById('open-diagnostico').addEventListener('click', ()=> openModal('modal-diagnostico'));
    document.getElementById('open-visita').addEventListener('click', ()=> openModal('modal-visita'));

    // close on backdrop click and ESC
    document.querySelectorAll('.modal-backdrop').forEach(back => {
      back.addEventListener('click', (e)=> { if(e.target === back) closeModal(back.id); })
    });
    document.addEventListener('keydown', (e)=> { if(e.key === 'Escape') document.querySelectorAll('.modal-backdrop').forEach(m=>closeModal(m.id)) });

    // ---------- FORM: Diagnóstico ----------
    document.getElementById('form-diagnostico').addEventListener('submit', function(e){
      e.preventDefault();
      const fd = new FormData(this);
      if(!fd.get('nombre')||!fd.get('email')||!fd.get('celular')) return showMsg('diagnostico-msg','error','Por favor complete los campos obligatorios.');
      let total=0;
      for(let i=1;i<=10;i++) total += Number(fd.get('q'+i) || 0);
      if(total === 100) total = 90;
      showMsg('diagnostico-msg','success',`Gracias ${fd.get('nombre')}. Puntaje: ${total}/100. Nuestro equipo te contactará.`);
      this.reset();
      // TODO: send to server via fetch()
    });

    // ---------- FORM: Visita ----------
    document.getElementById('form-visita').addEventListener('submit', function(e){
      e.preventDefault();
      const fd = new FormData(this);
      const required = ['nombre','email','celular','opt1','opt2','opt3','servicio'];
      for(const r of required) if(!fd.get(r)) return showMsg('visita-msg','error','Complete los campos obligatorios.');
      const o1 = fd.get('opt1'), o2 = fd.get('opt2'), o3 = fd.get('opt3');
      if(o1===o2||o1===o3||o2===o3) return showMsg('visita-msg','error','Seleccione tres fechas/hora distintas.');
      showMsg('visita-msg','success','Solicitud enviada. Le confirmaremos la visita al contacto indicado.');
      this.reset();
    });

    function showMsg(id,type,text){
      const el = document.getElementById(id);
      el.style.display = 'block';
      el.className = 'msg ' + (type==='success' ? 'success' : 'error');
      el.textContent = text;
      setTimeout(()=> el.style.opacity = '1', 10);
    }

    const observers = document.querySelectorAll('.fade-up');
    const io = new IntersectionObserver((entries, obs) => {
      entries.forEach(entry => {
        if(entry.isIntersecting){ entry.target.classList.add('appear'); obs.unobserve(entry.target); }
      });
    }, {threshold: .15});
    observers.forEach(o=>io.observe(o));
