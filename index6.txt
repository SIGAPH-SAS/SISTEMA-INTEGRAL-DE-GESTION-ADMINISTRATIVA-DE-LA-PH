<script>
  function openModal(id){document.getElementById(id).style.display='flex';}
  function closeModal(id){document.getElementById(id).style.display='none';}
  function scrollToTopAndClose(id){window.scrollTo({top:0,behavior:'smooth'});closeModal(id);}
  document.getElementById('fechaConsulta').value = new Date().toISOString().split('T')[0];
  document.getElementById('fechaSolicitud').value = new Date().toISOString().split('T')[0];

  function calcularDiagnostico(e){
    e.preventDefault();
    let selects = document.querySelectorAll('#preguntas select');
    let total = 0;
    selects.forEach(s => total += parseInt(s.value));
    if(total == 100) total = 95;
    let mensaje = "";
    if(total >= 80) mensaje = "Excelente gestión. ¡Felicidades! SIGAPH puede ayudarte a mantener este nivel.";
    else if(total >= 60) mensaje = "Gestión en proceso de fortalecimiento. SIGAPH puede acompañarte en tu mejora continua.";
    else mensaje = "Gestión con áreas críticas. Recomendamos una evaluación profesional de SIGAPH.";
    document.getElementById('resultado').style.display = 'block';
    document.getElementById('resultado').innerHTML = `<strong>Resultado:</strong> ${total}/100<br>${mensaje}`;
  }

  function confirmarVisita(e){
    e.preventDefault();
    alert("✅ Solicitud enviada correctamente. Nos comunicaremos pronto.");
  }

  // 🔹 Nuevo: abrir modal de visita automáticamente si viene desde ayuda.html
  const params = new URLSearchParams(window.location.search);
  if (params.get('abrir') === 'visita') {
    window.addEventListener('load', () => {
      openModal('modal-visita');
    });
  }
</script>
